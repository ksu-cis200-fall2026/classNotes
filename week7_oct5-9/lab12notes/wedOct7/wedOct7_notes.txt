Call-by-value vs call-by-reference

public static void zeroNum(int num) {
    num = 0;
}

public static void zeroArray(int[] nums) {
    for (int i = 0; i < nums.length; i++) {
        nums[i] = 0;
    }
}

int x = 4;
zeroNum(x);
System.out.println(x);

int[] stuff = {1,2,3};
zeroArray(stuff);
