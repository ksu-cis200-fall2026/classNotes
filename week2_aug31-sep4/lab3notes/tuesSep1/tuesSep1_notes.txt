        double price = num*5.99;
    }
    else {
        double price = num*3.99;
    }
    System.out.printf("Total is: $%.2f%n", price);

What is this trying to do? 

Will it compile? no, price is out of scope



Fix:

    double price;
    if (num < 3) {
        price = num*5.99;
    }
    else {
        price = num*3.99;
    }
    System.out.printf("Total is: $%.2f%n", price);
